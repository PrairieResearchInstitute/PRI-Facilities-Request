# Server environment file for PRI Facilities Request (drez-config-secrets, PHP track).
# One KEY= line per setting. The filled copy (pri-facilities-request-test.env,
# pri-facilities-request-live.env) sits beside compose.yaml on the server, readable
# only by the service account, and never enters git. Each tier has its own database.
#
# TIER=test or TIER=live switches the app onto this file (facilities/lib/serverenv.php):
# it then reads no config.php, no config.local.json and no pg.env, none of which the
# image carries. On this computer TIER stays unset and nothing here applies.
# Values marked NCSA come from NCSA when it defines the server; the move fills them in.

# Platform keys (config.template.json). TIER is test or live.
TIER=
DATA_FOLDER=
DATABASE_URL=
API_BASE_URL=

# Application (config.php on this computer). APP_ENVIRONMENT is production on both
# tiers [production]; APP_TIMEZONE [America/Chicago]; APP_KEY a new random value per tier.
APP_ENVIRONMENT=production
APP_TIMEZONE=
APP_KEY=

# Engine: postgres, the only engine on a server [postgres].
DATA_MODE=postgres

# PostgreSQL (pg.env on this computer). NCSA: host and port. PGDATABASE is
# facilities_test on the test tier and facilities_live on the live tier.
# The logins split: PGUSER is facilities_app (row rights only, used by the app);
# PG_OWNER_USER is facilities_owner, used only by tools/pg/migrate.php, which then
# grants facilities_app its row rights. Leave PG_OWNER_* out of the app's own
# env file if NCSA runs migrations from a separate job.
PGHOST=
PGPORT=
PGDATABASE=
PGUSER=
PGPASSWORD=
PG_OWNER_USER=
PG_OWNER_PASSWORD=
# TLS (Radiant audit M3): sslmode verify-full. compose.yaml mounts the NCSA CA
# file pg-ca.crt (copied beside compose.yaml) read-only at /run/secrets/pg-ca.crt.
# libpq, under pdo_pgsql, reads PGSSLROOTCERT for the CA file and, with sslmode
# verify-full, checks the certificate chain and the host name. Keep the value as
# written; libpq ignores a missing file unless the sslmode is verify-ca or verify-full.
PGSSLROOTCERT=/run/secrets/pg-ca.crt
# tools/pg/migrate.php runs through docker compose run and gets the same mount.

# NetID sign-in (facilities/lib/identity.php, docs/decisions.md D-014). The server
# tiers set AUTH_MODE=proxy: the university's authenticating proxy passes the NetID
# in a trusted header. Blank values take the defaults shown in brackets.
# AUTH_MODE                 proxy on the server; local [default] is this computer only
# AUTH_USER_HEADER          NCSA: header holding the NetID or NetID@domain [X-Forwarded-Email]
# AUTH_EMAIL_HEADER         NCSA: header holding the address [X-Forwarded-Email]
# AUTH_NAME_HEADER          NCSA: header holding the display name [none]
# AUTH_GROUPS_HEADER        NCSA: header listing the campus groups [X-Forwarded-Groups]
# AUTH_GROUPS_SEPARATOR     NCSA: character between groups [,]
# AUTH_NETID_DOMAINS        comma list of domains whose addresses are NetIDs [illinois.edu]
# AUTH_TRUSTED_PROXIES      NCSA: comma list of proxy addresses or CIDR ranges; any other sender is refused [none: trust the network]
# AUTH_ALLOWED_GROUPS       comma list; when set, only members of one of these groups may sign in [none: any NetID, as staff]
# AUTH_LOGOUT_URL           NCSA: the proxy's sign-out address, for the Sign out link [none]
AUTH_MODE=proxy
AUTH_USER_HEADER=
AUTH_EMAIL_HEADER=
AUTH_NAME_HEADER=
AUTH_GROUPS_HEADER=
AUTH_GROUPS_SEPARATOR=
AUTH_NETID_DOMAINS=
AUTH_TRUSTED_PROXIES=
AUTH_ALLOWED_GROUPS=
AUTH_LOGOUT_URL=
